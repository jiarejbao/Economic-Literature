% ------------------------- 自定义盒子 -------------------------
% 三类盒子按章编号 (识别要点 2.1、易错点 2.3 …), 三类计数器各自独立。
% 需被跨章引用的盒子, 在盒内首行放 \label{pit:xxx}, 引用处写「第 2 章的易错点~\ref{pit:xxx}」。
% 识别要点 (可整体搬用的模型设定、口径与识别设计)
\newtcolorbox[auto counter, number within=chapter]{ident}[1][]{colback=green!4,
  colframe=green!45!black, title={\textbf{识别要点~\thetcbcounter}~#1},
  fonttitle=\small, left=2mm, right=2mm, top=1mm, bottom=1mm, breakable}
% 易错点
\newtcolorbox[auto counter, number within=chapter]{pitfall}[1][]{colback=red!4,
  colframe=red!60!black, title={\textbf{易错点~\thetcbcounter}~#1},
  fonttitle=\small, left=2mm, right=2mm, top=1mm, bottom=1mm, breakable}
% 直觉/为什么
\newtcolorbox[auto counter, number within=chapter]{intuition}[1][]{colback=blue!4,
  colframe=blue!50!black, title={\textbf{直觉~\thetcbcounter}~#1},
  fonttitle=\small, left=2mm, right=2mm, top=1mm, bottom=1mm, breakable}
% 原文对照(备用: 体例是页码/表号随正文与盒标题就近标注, 本盒一般不用)
\newtcolorbox{origin}[1][]{colback=black!3, colframe=black!40,
  title={\textbf{原文出处}~#1}, fonttitle=\small, left=2mm, right=2mm,
  top=1mm, bottom=1mm, breakable}
